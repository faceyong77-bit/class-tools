  <script>
    const STORAGE_KEY = 'via-field-trip-report-v3';
    const fields = ['studentName', 'place', 'date', 'value', 'photoDesc', 'question', 'found', 'review', 'action'];
    const checks = ['checkReason', 'checkQuestion', 'checkAction'];

    function byId(id) {
      return document.getElementById(id);
    }

    function getValue(id) {
      const el = byId(id);
      return el ? el.value : '';
    }

    function setValue(id, value) {
      const el = byId(id);
      if (el) el.value = value || '';
    }

    function getChecked(id) {
      const el = byId(id);
      return !!(el && el.checked);
    }

    function setChecked(id, value) {
      const el = byId(id);
      if (el) el.checked = !!value;
    }

    function loadPhoto(event) {
      const file = event.target.files && event.target.files[0];
      if (!file) return;

      const reader = new FileReader();
      reader.onload = function(e) {
        const img = byId('preview');
        const placeholder = byId('photoPlaceholder');
        if (!img || !placeholder) return;

        img.src = e.target.result;
        img.classList.remove('hidden');
        placeholder.classList.add('hidden');
        localStorage.setItem(STORAGE_KEY + '-photo', e.target.result);
      };
      reader.readAsDataURL(file);
    }

    function saveToLocal() {
      const data = {};
      fields.forEach(id => data[id] = getValue(id));
      checks.forEach(id => data[id] = getChecked(id));
      localStorage.setItem(STORAGE_KEY, JSON.stringify(data));
    }

    function loadFromLocal() {
      const saved = localStorage.getItem(STORAGE_KEY);
      if (saved) {
        try {
          const data = JSON.parse(saved);
          fields.forEach(id => setValue(id, data[id]));
          checks.forEach(id => setChecked(id, data[id]));
        } catch (error) {
          console.warn('저장된 자료를 불러오지 못했습니다.', error);
        }
      }

      const photo = localStorage.getItem(STORAGE_KEY + '-photo');
      if (photo) {
        const img = byId('preview');
        const placeholder = byId('photoPlaceholder');
        if (img && placeholder) {
          img.src = photo;
          img.classList.remove('hidden');
          placeholder.classList.add('hidden');
        }
      }
    }

    document.addEventListener('input', function(e) {
      if (e.target && fields.includes(e.target.id)) saveToLocal();
    });

    document.addEventListener('change', function(e) {
      if (e.target && checks.includes(e.target.id)) saveToLocal();
    });

    function getSafeFileName(text) {
      return String(text || '')
        .replace(/[^가-힣a-zA-Z0-9_\-]/g, '_')
        .replace(/_+/g, '_')
        .slice(0, 40) || '답사보고서';
    }

    function makeStaticReportClone() {
      const original = byId('report');
      if (!original) throw new Error('보고서 영역을 찾을 수 없습니다.');

      const clone = original.cloneNode(true);

      clone.querySelectorAll('input, textarea').forEach(el => {
        if (el.type === 'checkbox') {
          const mark = document.createElement('span');
          mark.textContent = el.checked ? '☑' : '☐';
          mark.style.fontWeight = '800';
          mark.style.marginRight = '6px';
          el.replaceWith(mark);
          return;
        }

        const box = document.createElement('div');
        box.className = el.className;
        box.style.whiteSpace = 'pre-wrap';
        box.style.minHeight = el.tagName === 'TEXTAREA' ? '80px' : '48px';
        box.style.background = '#ffffff';
        box.style.color = '#1e293b';
        box.style.border = '1px solid #cbd5e1';
        box.style.borderRadius = '14px';
        box.style.padding = '12px';
        box.textContent = el.value || '';
        el.replaceWith(box);
      });

      clone.querySelectorAll('.no-print').forEach(el => el.remove());
      clone.style.width = '210mm';
      clone.style.minHeight = '297mm';
      clone.style.boxShadow = 'none';
      clone.style.margin = '0';
      clone.style.borderRadius = '0';
      clone.style.background = '#ffffff';
      return clone;
    }

    function savePDF() {
      saveToLocal();

      if (typeof html2pdf === 'undefined') {
        alert('직접 PDF 저장 기능을 불러오지 못했습니다. 인쇄창에서 PDF로 저장해 주세요.');
        printPDF();
        return;
      }

      const name = getSafeFileName(getValue('studentName').trim() || '학생');
      const place = getSafeFileName(getValue('place').trim() || '답사보고서');
      const clone = makeStaticReportClone();
      const temp = document.createElement('div');
      temp.style.position = 'fixed';
      temp.style.left = '-10000px';
      temp.style.top = '0';
      temp.style.width = '210mm';
      temp.style.background = '#ffffff';
      temp.appendChild(clone);
      document.body.appendChild(temp);

      const opt = {
        margin: [5, 5, 5, 5],
        filename: `${name}_${place}_VIA답사보고서.pdf`,
        image: { type: 'jpeg', quality: 0.98 },
        html2canvas: { scale: 2, useCORS: true, allowTaint: true, backgroundColor: '#ffffff', scrollY: 0 },
        jsPDF: { unit: 'mm', format: 'a4', orientation: 'portrait' },
        pagebreak: { mode: ['css', 'legacy'] }
      };

      html2pdf().set(opt).from(clone).save().then(function() {
        if (temp.parentNode) document.body.removeChild(temp);
      }).catch(function(error) {
        if (temp.parentNode) document.body.removeChild(temp);
        console.error('PDF 저장 오류:', error);
        alert('직접 PDF 저장이 막혔습니다. 인쇄창에서 대상을 PDF로 저장으로 선택해 주세요.');
        printPDF();
      });
    }

    function printPDF() {
      saveToLocal();
      window.print();
    }

    function clearAll() {
      if (!confirm('입력한 내용을 모두 지울까요?')) return;
      fields.forEach(id => setValue(id, ''));
      checks.forEach(id => setChecked(id, false));

      const img = byId('preview');
      const placeholder = byId('photoPlaceholder');
      if (img) {
        img.src = '';
        img.classList.add('hidden');
      }
      if (placeholder) placeholder.classList.remove('hidden');

      localStorage.removeItem(STORAGE_KEY);
      localStorage.removeItem(STORAGE_KEY + '-photo');
    }

    function runSelfTests() {
      const requiredIds = ['report', 'studentName', 'place', 'date', 'value', 'photoDesc', 'question', 'found', 'review', 'action', 'preview', 'photoPlaceholder'];
      const missing = requiredIds.filter(id => !byId(id));
      console.assert(missing.length === 0, '필수 요소 누락:', missing);

      console.assert(getSafeFileName('홍길동/좌동:유적?') === '홍길동_좌동_유적_', '파일명 안전 처리 테스트 실패');

      const before = getValue('value');
      setValue('value', '테스트 문장');
      console.assert(getValue('value') === '테스트 문장', '입력값 저장/읽기 테스트 실패');
      setValue('value', before);
    }

    loadFromLocal();
    runSelfTests();
  </script>
